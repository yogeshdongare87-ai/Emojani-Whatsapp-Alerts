import pandas as pd import pyautogui import urllib.parse import time import webbrowser import tkinter as tk from tkinter import filedialog def main(): print("📂 Step 1: Apni 'contacts.xlsx' file select karein...") root = tk.Tk() root.withdraw() contacts_path = filedialog.askopenfilename(title="Contacts Excel File Chunein", filetypes=[("Excel Files", "*.xlsx *.xls")]) if not contacts_path: print("❌ Contacts file select nahi ki gayi.") return print(f"✓ Contacts File: {contacts_path}") print("📂 Step 2: Apni 'Report_7' Excel file select karein...") report_path = filedialog.askopenfilename(title="Report Excel File Chunein", filetypes=[("Excel Files", "*.xlsx *.xls")]) if not report_path: print("❌ Report file select nahi ki gayi.") return print(f"✓ Report File: {report_path}") # Contacts Data Load Karein df_contacts = pd.read_excel(contacts_path) contacts_by_name = {} contacts_by_taluka_status = {} for idx, row in df_contacts.iterrows(): name = str(row.iloc[0]).strip() if pd.notna(row.iloc[0]) else "" taluka = str(row.iloc[1]).strip() if len(row) > 1 and pd.notna(row.iloc[1]) else "" status = str(row.iloc[2]).strip() if len(row) > 2 and pd.notna(row.iloc[2]) else "" mob = "" for val in row.values: s = ''.join(filter(str.isdigit, str(val))) if len(s) >= 10: mob = s[-10:] break if name and mob: contacts_by_name[name.lower()] = {"name": name, "mob": mob} if taluka and status and mob: key = f"{taluka}___{status}".lower() contacts_by_taluka_status[key] = {"name": name if name else status, "mob": mob} # Report Data Load aur Process Karein df_report = pd.read_excel(report_path, header=None) data_rows = df_report.iloc[4:].values grouped = {} for row in data_rows: if len(row) < 9: continue taluka = str(row[2]).strip() if pd.notna(row[2]) else "" surveyor = str(row[4]).strip() if pd.notna(row[4]) else "" status = str(row[5]).strip() if pd.notna(row[5]) else "" try: days = int(row[8]) except: continue if not status or status == 'स्थिती': continue assigned_name = "" role_tag = "" target_mob = "" if "मोजणीची माहिती" in status: role_tag = "भूकरमापक" assigned_name = surveyor if surveyor.lower() in contacts_by_name: target_mob = contacts_by_name[surveyor.lower()]["mob"] else: combo_key = f"{taluka}___{status}".lower() matched = contacts_by_taluka_status.get(combo_key) if "छाननी लिपिक" in status: role_tag = "छाननी लिपिक" elif "शिरस्तेदार" in status or "मुख्यालय" in status: role_tag = "शिरस्तेदार" elif "ऊप.अ" in status or "मान्यतेवर" in status or "भू. अ" in status: role_tag = "उपभूमी अभिलेख अधिकारी" else: role_tag = "अधिकारी" if matched: assigned_name = matched["name"] target_mob = matched["mob"] else: assigned_name = f"{role_tag} ({taluka})" target_mob = "" if not assigned_name or assigned_name == 'nan': continue group_key = f"{assigned_name}___{taluka}___{role_tag}" if group_key not in grouped: grouped[group_key] = { "name": assigned_name, "taluka": taluka, "role": role_tag, "mob": target_mob, "total": 0, "d180": 0, "d90_180": 0, "d60_90": 0, "d30_60": 0, "d15_30": 0, "d0_15": 0 } if not grouped[group_key]["mob"] and target_mob: grouped[group_key]["mob"] = target_mob grouped[group_key]["total"] += 1 # Time Buckets logic if days > 180: grouped[group_key]["d180"] += 1 elif days > 90: grouped[group_key]["d90_180"] += 1 elif 60 <= days <= 90: grouped[group_key]["d60_90"] += 1 elif 30 <= days < 60: grouped[group_key]["d30_60"] += 1 elif 15 <= days < 30: grouped[group_key]["d15_30"] += 1 elif days < 15: grouped[group_key]["d0_15"] += 1 print(f"\n✓ Kul {len(grouped)} adhikari/roles ke pralambit cases mil gaye hain.") print("🚀 WhatsApp Automation 5 sekund mein shuru ho raha hai...") time.sleep(5) sent_count = 0 for key, info in grouped.items(): mob = info["mob"] if not mob or len(mob) < 10: print(f"⚠️ Number nahi mila: {info['name']} ({info['role']} - {info['taluka']})") continue msg = ( f"नमस्कार {info['name']} सर/मॅडम,\n\n" f"भूमी अभिलेख विभाग - प्रलंबित प्रकरणे सूचना:\n" f"आपल्याकडे ({info['role']} - {info['taluka']}) एकूण {info['total']} प्रकरणे प्रलंबित आहेत:\n" f"• १८० दिवसांपेक्षा जास्त: {info['d180']} केस\n" f"• ९० ते १८० दिवस: {info['d90_180']} केस\n" f"• ६० ते ९० दिवस: {info['d60_90']} केस\n" f"• ३० ते ६० दिवस: {info['d30_60']} केस\n" f"• १५ ते ३० दिवस: {info['d15_30']} केस\n" f"• ० ते १५ दिवस: {info['d0_15']} केस\n\n" f"तरी आपणास नम्र विनंती आहे की, कृपया प्रलंबित प्रकरणांचा आढावा घेऊन त्यांचा शक्य तितक्या लवकर निपटारा करण्यासाठी आवश्यक कार्यवाही करावी. आपल्या सहकार्याबद्दल मनःपूर्वक धन्यवाद." ) encoded_msg = urllib.parse.quote(msg) url = f"https://web.whatsapp.com/send?phone=+91{mob}&text={encoded_msg}" webbrowser.open(url) time.sleep(13) pyautogui.press('enter') sent_count += 1 print(f"[{sent_count}] ✅ Message sent to {info['name']} ({mob}) - Total Cases: {info['total']}") time.sleep(5) print(f"\n🎉 Bahut badhiya! Kul {sent_count} adhikariyon ko WhatsApp sandesh bhej diye gaye hain.") if __name__ == "__main__": main()
